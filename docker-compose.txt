# YMB gateway topology: only NGINX is mapped to a host port.
services:
  mongo:
    image: mongo:6
    restart: unless-stopped
    volumes:
      - mongo_data:/data/db
    networks: [data]
  backend:
    build:
      context: ./backend
      dockerfile: Dockerfile
    environment:
      MONGO_URI: mongodb://mongo:27017/bank_app
      SECRET_KEY: local-demo-change-this
    depends_on: [mongo]
    networks: [app, data]
  transactions:
    build:
      context: ./transactions
      dockerfile: Dockerfile
    environment:
      MONGO_URI: mongodb://mongo:27017
      PORT: "3000"
    depends_on: [mongo]
    networks: [app, data]
  frontend:
    build:
      context: ./studentportfolio
      dockerfile: Dockerfile
    networks: [app]
  nginx:
    image: nginx:alpine
    ports: ["80:80"]
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on: [frontend, backend, transactions]
    networks: [app]
volumes:
  mongo_data:
networks:
  app:
  data:
